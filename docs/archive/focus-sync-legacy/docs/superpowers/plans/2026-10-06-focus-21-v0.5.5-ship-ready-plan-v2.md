```rust
            hwnd_u32 as isize,
            &current,
            false,
        );
        // ... log Write / WriteFailed ...
    } else {
        // v0.5.5: defensive log when Fix B guard kicks in.
        // Helps diagnose "why dialog didn't get my path" — answer is
        // "extension never PUSHed yet, please wait or click Sync Now".
        log_event(&Event::SpikeError {
            kind: "opened_write_back_skip_sentinel".into(),
            message: format!(
                "dialog path='{}' current='{}' (sentinel); skipping until first push received",
                initial_path, current
            ),
            ts: now_iso8601(),
        });
    }
}
```
